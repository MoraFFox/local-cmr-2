## 1. Supabase Table Design

### `logistics_operations` Table

```sql
CREATE TABLE logistics_operations (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  customer_id BIGINT NOT NULL,  -- references company/submission
  machine_id BIGINT,            -- optional: specific customer machine
  opened_by_record_id BIGINT,   -- maintenance record that opened this
  closed_by_record_id BIGINT,   -- maintenance record that closed this (nullable until closed)
  operation_type TEXT NOT NULL, -- 'pickup_and_deliver' | 'deliver_only' | 'pickup_only'
  status TEXT NOT NULL DEFAULT 'open', -- 'open' | 'closed'
  machine_category TEXT,        -- 'coffee' | 'grinder' | 'other'
  machine_ownership TEXT,       -- 'customer' | 'company'
  machine_type TEXT,            -- 'manual' | 'automatic' | 'semi_automatic'
  replacement_machine_id BIGINT REFERENCES company_machines(id),
  monthly_rental_price NUMERIC(10,2),
  daily_rental_price NUMERIC(10,2) GENERATED ALWAYS AS (monthly_rental_price / 30) STORED,
  rental_duration_days INTEGER,
  rental_duration_hours INTEGER,
  rental_duration_minutes INTEGER,
  billable_days INTEGER,
  total_rental_cost NUMERIC(10,2),
  pickup_cost NUMERIC(10,2) DEFAULT 0,
  return_cost NUMERIC(10,2) DEFAULT 0,
  total_logistics_cost NUMERIC(10,2) GENERATED ALWAYS AS (
    COALESCE(total_rental_cost, 0) + COALESCE(pickup_cost, 0) + COALESCE(return_cost, 0)
  ) STORED,
  internal_notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE logistics_operations ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Authenticated users can read logistics_operations"
  ON logistics_operations FOR SELECT
  USING (auth.role() = 'authenticated');

CREATE POLICY "Authenticated users can insert logistics_operations"
  ON logistics_operations FOR INSERT
  WITH CHECK (auth.role() = 'authenticated');

CREATE POLICY "Authenticated users can update logistics_operations"
  ON logistics_operations FOR UPDATE
  USING (auth.role() = 'authenticated');
```

**Key decisions:**
- `daily_rental_price` is a **generated column** — computed automatically from `monthly_rental_price / 30`, no code needed
- `total_logistics_cost` is a **generated column** — auto-sums rental + pickup + return
- Rental duration stored as 3 separate integer fields (days, hours, minutes) for billing flexibility
- `status` starts as `'open'`, becomes `'closed'` when a closing record is linked

### `company_machines` Table

```sql
CREATE TABLE company_machines (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,
  category TEXT NOT NULL,        -- 'coffee' | 'grinder' | 'other'
  machine_type TEXT,             -- 'manual' | 'automatic' | 'semi_automatic'
  status TEXT DEFAULT 'available', -- 'available' | 'in_use' | 'damaged' | 'lost'
  monthly_rental_price NUMERIC(10,2),
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE company_machines ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Authenticated users can read company_machines"
  ON company_machines FOR SELECT
  USING (auth.role() = 'authenticated');

CREATE POLICY "Authenticated users can manage company_machines"
  ON company_machines FOR ALL
  USING (auth.role() = 'authenticated');
```

**Migration pattern:** Follow existing convention — create migration file at `supabase/migrations/YYYYMMDDHHMMSS_create_logistics_tables.sql`.

### Supabase-js Query Patterns

**Fetch open operations for a customer:**
```ts
const { data } = await supabase
  .from('logistics_operations')
  .select('*, company_machines(*)')
  .eq('customer_id', customerId)
  .eq('status', 'open')
  .order('created_at', { ascending: false });
```

**Close an operation:**
```ts
const { data } = await supabase
  .from('logistics_operations')
  .update({
    status: 'closed',

    billable_days: billableDays,
    total_rental_cost: rentalCost,

  })
  .eq('id', operationId)
  .select();
```

## 4. Custom Hook Pattern for Logistics Data

Following the existing `useCustomCatalog` pattern:

```ts
// hooks/useLogisticsOperations.ts
export function useLogisticsOperations(customerId: number | null) {
  const [operations, setOperations] = useState<LogisticsOperation[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const fetchOperations = useCallback(async () => {
    if (!customerId) return;
    try {
      const { data, error } = await supabase
        .from('logistics_operations')
        .select('*, company_machines(*)')
        .eq('customer_id', customerId)
        .order('created_at', { ascending: false });
      if (error) throw error;
      setOperations(data || []);
    } catch (err) {
      setError(err instanceof Error ? err : new Error(String(err)));
    } finally {
      setIsLoading(false);
    }
  }, [customerId]);

  useEffect(() => { fetchOperations(); }, [fetchOperations]);

  const createOperation = async (op: CreateLogisticsOperation) => { ... };
  const closeOperation = async (id: number, closeData: CloseOperationData) => { ... };

  return { operations, isLoading, error, createOperation, closeOperation, refresh: fetchOperations };
}
```

Self-contained hook — no centralized state management needed (addresses CONCERNS.md #3).

## 7. Date Calculation Strategy

**Critical rule:** Use maintenance record `maintenanceDate` fields, never `new Date()` or `created_at`.

```ts
function calculateRentalDuration(
  openRecordDate: string,   // from opened_by_record's maintenanceDate
  closeRecordDate: string,  // from closed_by_record's maintenanceDate
): { days: number; hours: number; minutes: number } {
  const start = new Date(openRecordDate);
  const end = new Date(closeRecordDate);
  const diffMs = end.getTime() - start.getTime();
  const totalMinutes = Math.floor(diffMs / 60000);
  return {
    days: Math.floor(totalMinutes / 1440),
    hours: Math.floor((totalMinutes % 1440) / 60),
    minutes: totalMinutes % 60,
  };
}

function calculateBillableDays(duration: { days: number }): number {
  return duration.days; // Simple policy: billable = calendar days
}

function calculateDailyRentalPrice(monthlyPrice: number): number {
  return Math.round((monthlyPrice / 30) * 100) / 100;
}
```
